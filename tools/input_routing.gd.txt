# Gameplay touch ownership is independent of Godot's single emulated mouse.
var gameplay_touches:Dictionary={}

func _reset_gameplay_input()->void:
	movement=Vector2.ZERO
	if is_instance_valid(joystick):joystick.reset_input()
	for button in gameplay_touches.values():
		if is_instance_valid(button):button.set_pressed_no_signal(false)
	gameplay_touches.clear()

func _gameplay_button_at(point:Vector2)->Button:
	if not is_instance_valid(hud) or not hud.is_visible_in_tree():return null
	# A modal is the only eligible input root while open, including its close
	# controls. Never let rectangle hit testing reach through its background.
	var input_root:Control=popup if is_instance_valid(popup) else hud
	return _touch_button_in(input_root,point)

func _touch_button_in(control:Node,point:Vector2)->Button:
	var children:Array=control.get_children()
	children.reverse()
	for child in children:
		if child is Control and child.is_visible_in_tree() and not child.is_queued_for_deletion():
			var nested:Button=_touch_button_in(child,point)
			if nested!=null:return nested
			if child is Button and not child.disabled and child.get_global_rect().has_point(point):return child
	return null

func _input(event:InputEvent)->void:
	# Physical mouse/keyboard continue through the native Button implementation.
	# Suppress only the duplicate mouse events synthesized from touchscreen input.
	if event is InputEventMouse and event.device==InputEvent.DEVICE_ID_EMULATION:
		if _gameplay_button_at(event.position)!=null or not gameplay_touches.is_empty():
			get_viewport().set_input_as_handled()
		return
	if event is InputEventScreenTouch:
		if event.pressed and not event.canceled:
			var button:Button=_gameplay_button_at(event.position)
			if button!=null:
				# One button activation per touch; a second finger cannot steal it.
				if not gameplay_touches.values().has(button):
					gameplay_touches[event.index]=button
					button.set_pressed_no_signal(true)
				get_viewport().set_input_as_handled()
		elif gameplay_touches.has(event.index):
			var button:Button=gameplay_touches[event.index]
			gameplay_touches.erase(event.index)
			if is_instance_valid(button):
				button.set_pressed_no_signal(false)
				if not event.canceled and _gameplay_button_at(event.position)==button:
					button.pressed.emit()
			get_viewport().set_input_as_handled()
	elif event is InputEventScreenDrag and gameplay_touches.has(event.index):
		var button:Button=gameplay_touches[event.index]
		if is_instance_valid(button):button.set_pressed_no_signal(_gameplay_button_at(event.position)==button)
		get_viewport().set_input_as_handled()
